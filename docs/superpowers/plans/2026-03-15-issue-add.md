// ABOUTME: Git ref storage operations for chain entities (issues, milestones, config).
// ABOUTME: Wraps go-git to create blobs, trees, and commits on per-entity refs.
package storage

import (
	"fmt"
	"strings"
	"time"

	git "github.com/go-git/go-git/v5"
	"github.com/go-git/go-git/v5/config"
	"github.com/go-git/go-git/v5/plumbing"
	"github.com/go-git/go-git/v5/plumbing/object"
	"github.com/go-git/go-git/v5/plumbing/storer"
)

// Store provides read/write access to chain entity refs in a git repository.
type Store struct {
	repo   *git.Repository
	author object.Signature
}

// NewStore creates a Store backed by the given git repository.
// Reads user.name and user.email from git config for commit authoring.
func NewStore(repo *git.Repository) (*Store, error) {
	cfg, err := repo.ConfigScoped(config.GlobalScope)
	if err != nil {
		cfg = &config.Config{}
	}

	name := cfg.User.Name
	email := cfg.User.Email
	if name == "" {
		name = "git-chain"
	}
	if email == "" {
		email = "git-chain@local"
	}

	return &Store{
		repo: repo,
		author: object.Signature{
			Name:  name,
			Email: email,
		},
	}, nil
}

// WriteEntity creates or appends a commit on the given ref containing
// the specified file with the given content. If the ref already exists,
// the new commit's parent is the current ref target.
func (s *Store) WriteEntity(refPath, filename string, content []byte, message string) error {
	storer := s.repo.Storer

	// Create blob
	blobObj := plumbing.MemoryObject{}
	blobObj.SetType(plumbing.BlobObject)
	blobObj.SetSize(int64(len(content)))
	w, err := blobObj.Writer()
	if err != nil {
		return fmt.Errorf("create blob writer: %w", err)
	}
	if _, err := w.Write(content); err != nil {
		return fmt.Errorf("write blob: %w", err)
	}
	if err := w.Close(); err != nil {
		return fmt.Errorf("close blob: %w", err)
	}
	blobHash, err := storer.SetEncodedObject(&blobObj)
	if err != nil {
		return fmt.Errorf("store blob: %w", err)
	}

	// Create tree with one entry
	tree := object.Tree{
		Entries: []object.TreeEntry{
			{
				Name: filename,
				Mode: 0100644,
				Hash: blobHash,
			},
		},
	}
	treeObj := plumbing.MemoryObject{}
	if err := tree.Encode(&treeObj); err != nil {
		return fmt.Errorf("encode tree: %w", err)
	}
	treeHash, err := storer.SetEncodedObject(&treeObj)
	if err != nil {
		return fmt.Errorf("store tree: %w", err)
	}

	// Build commit
	now := time.Now()
	commit := object.Commit{
		Author: object.Signature{
			Name:  s.author.Name,
			Email: s.author.Email,
			When:  now,
		},
		Committer: object.Signature{
			Name:  s.author.Name,
			Email: s.author.Email,
			When:  now,
		},
		Message:  message,
		TreeHash: treeHash,
	}

	// If ref exists, set parent to current commit
	ref, err := storer.Reference(plumbing.ReferenceName(refPath))
	if err == nil {
		commit.ParentHashes = []plumbing.Hash{ref.Hash()}
	}

	commitObj := plumbing.MemoryObject{}
	if err := commit.Encode(&commitObj); err != nil {
		return fmt.Errorf("encode commit: %w", err)
	}
	commitHash, err := storer.SetEncodedObject(&commitObj)
	if err != nil {
		return fmt.Errorf("store commit: %w", err)
	}

	// Update or create ref
	newRef := plumbing.NewHashReference(plumbing.ReferenceName(refPath), commitHash)
	if err := storer.SetReference(newRef); err != nil {
		return fmt.Errorf("set reference: %w", err)
	}

	return nil
}

// ReadEntity reads the named file from the latest commit on the given ref.
func (s *Store) ReadEntity(refPath, filename string) ([]byte, error) {
	ref, err := s.repo.Storer.Reference(plumbing.ReferenceName(refPath))
	if err != nil {
		return nil, fmt.Errorf("ref %s not found: %w", refPath, err)
	}

	commitObj, err := object.GetCommit(s.repo.Storer, ref.Hash())
	if err != nil {
		return nil, fmt.Errorf("get commit: %w", err)
	}

	tree, err := commitObj.Tree()
	if err != nil {
		return nil, fmt.Errorf("get tree: %w", err)
	}

	entry, err := tree.FindEntry(filename)
	if err != nil {
		return nil, fmt.Errorf("file %s not found in tree: %w", filename, err)
	}

	blob, err := object.GetBlob(s.repo.Storer, entry.Hash)
	if err != nil {
		return nil, fmt.Errorf("get blob: %w", err)
	}

	reader, err := blob.Reader()
	if err != nil {
		return nil, fmt.Errorf("read blob: %w", err)
	}
	defer reader.Close()

	buf := make([]byte, blob.Size)
	if _, err := reader.Read(buf); err != nil && err.Error() != "EOF" {
		return nil, fmt.Errorf("read blob content: %w", err)
	}

	return buf, nil
}

// ListRefs returns all ref names that start with the given prefix.
func (s *Store) ListRefs(prefix string) ([]string, error) {
	iter, err := s.repo.Storer.IterReferences()
	if err != nil {
		return nil, fmt.Errorf("iterate refs: %w", err)
	}

	var refs []string
	err = iter.ForEach(func(ref *plumbing.Reference) error {
		name := string(ref.Name())
		if strings.HasPrefix(name, prefix) {
			refs = append(refs, name)
		}
		return nil
	})
	if err != nil {
		return nil, fmt.Errorf("list refs: %w", err)
	}

	return refs, nil
}

// RefExists checks whether the given ref exists.
func (s *Store) RefExists(refPath string) bool {
	_, err := s.repo.Storer.Reference(plumbing.ReferenceName(refPath))
	return err == nil
}

// Parse splits raw markdown into YAML frontmatter and body, then
// unmarshals the frontmatter into an Issue. The ID field is not in
// the frontmatter — the caller sets it from the ref path.
func Parse(raw []byte) (*Issue, error) {
	var iss Issue
	rest, err := frontmatter.Parse(bytes.NewReader(raw), &iss, frontmatter.NewFormat("---", "---", yaml.Unmarshal))
	if err != nil {
		return nil, fmt.Errorf("parse frontmatter: %w", err)
	}

	// frontmatter.Parse returns the body as []byte, not io.Reader
	iss.Body = strings.TrimSpace(string(rest))

	return &iss, nil
}

// Marshal serializes an Issue back to YAML frontmatter + markdown body.
func Marshal(iss *Issue) ([]byte, error) {
	frontmatterBytes, err := yaml.Marshal(iss)
	if err != nil {
		return nil, fmt.Errorf("marshal frontmatter: %w", err)
	}

	var buf bytes.Buffer
	buf.WriteString("---\n")
	buf.Write(frontmatterBytes)
	buf.WriteString("---\n")
	if iss.Body != "" {
		buf.WriteString("\n")
		buf.WriteString(iss.Body)
		buf.WriteString("\n")
	}

	return buf.Bytes(), nil
}

// SplitBatch splits a multi-issue input (separated by ---) into
// individual issue blocks. Each block includes its own frontmatter.
// Issues are separated by a --- line that is followed by YAML key-value
// pairs (indicating a new issue's frontmatter), as opposed to the ---
// that closes the current issue's frontmatter.
func SplitBatch(raw []byte) [][]byte {
	if len(bytes.TrimSpace(raw)) == 0 {
		return nil
	}

	text := string(raw)
	var blocks [][]byte

	// Strategy: find --- lines that start a NEW issue's frontmatter.
	// The first --- is always the opening of the first issue.
	// A subsequent --- that is FOLLOWED by a YAML key (word + colon)
	// on the next line starts a new issue.
	lines := strings.Split(text, "\n")
	var current []string
	inFrontmatter := false
	frontmatterSeen := false

	for i := 0; i < len(lines); i++ {
		trimmed := strings.TrimSpace(lines[i])

		if trimmed == "---" {
			if !frontmatterSeen {
				// Opening --- of first issue
				inFrontmatter = true
				frontmatterSeen = true
				current = append(current, lines[i])
			} else if inFrontmatter {
				// Closing --- of current frontmatter
				inFrontmatter = false
				current = append(current, lines[i])
			} else {
				// A --- in the body area. Check if the next non-empty line
				// looks like YAML (contains a colon), indicating a new issue.
				isNewIssue := false
				for j := i + 1; j < len(lines); j++ {
					nextTrimmed := strings.TrimSpace(lines[j])
					if nextTrimmed == "" {
						continue
					}
					// A YAML key looks like "word:" or "word: value" at the start of the line.
				// Reject lines where the colon is not preceded by a simple key name
				// (e.g., URLs like https://example.com or prose like "Note: something").
				if len(nextTrimmed) > 0 && isYAMLKey(nextTrimmed) {
						isNewIssue = true
					}
					break
				}

				if isNewIssue {
					// Save current block, start new one
					blocks = append(blocks, []byte(strings.Join(current, "\n")))
					current = []string{lines[i]}
					inFrontmatter = true
				} else {
					current = append(current, lines[i])
				}
			}
		} else {
			current = append(current, lines[i])
		}
	}

	if len(current) > 0 {
		blocks = append(blocks, []byte(strings.Join(current, "\n")))
	}

	return blocks
}

// isYAMLKey returns true if the line looks like a YAML key-value pair.
// Matches patterns like "title: value", "state: pending", "title:" (key only).
// Rejects prose like "Note: something" or URLs like "https://example.com".
func isYAMLKey(line string) bool {
	// YAML keys are simple identifiers: lowercase letters, digits, underscores
	idx := strings.Index(line, ":")
	if idx <= 0 {
		return false
	}
	key := line[:idx]
	for _, c := range key {
		if !((c >= 'a' && c <= 'z') || (c >= '0' && c <= '9') || c == '_') {
			return false
		}
	}
	return true
}

PersistentPreRunE: func(cmd *cobra.Command, args []string) error {
    // If an App was already injected (e.g., by test harness), keep it.
    if GetApp(cmd.Context()) != nil {

    }

    }

    return nil

func newIssueAddCommand() *cobra.Command {
	cmd := &cobra.Command{
		Use:   "add",
		Short: "Create one or more new issues",
		RunE:  runIssueAdd,
	}
	cmd.Flags().String("milestone", "", "assign to specific milestone")
	cmd.Flags().String("after", "", "position after another issue (not yet implemented)")
	cmd.Flags().String("before", "", "position before another issue (not yet implemented)")
	return cmd
}

func runIssueAdd(cmd *cobra.Command, args []string) error {
	app := GetApp(cmd.Context())
	if app == nil {
		return fmt.Errorf("not in a git repository")
	}

	if err := app.EnsureInitialized(); err != nil {
		return fmt.Errorf("initialize chain: %w", err)
	}

	// Check for not-yet-implemented flags
	if after, _ := cmd.Flags().GetString("after"); after != "" {
		return fmt.Errorf("--after is not yet implemented")
	}
	if before, _ := cmd.Flags().GetString("before"); before != "" {
		return fmt.Errorf("--before is not yet implemented")
	}

	// Read default milestone from config
	cfgData, err := app.Store.ReadEntity("refs/chain/_/config", "config.yaml")
	if err != nil {
		return fmt.Errorf("read config: %w", err)
	}
	var cfg config.Config
	if err := yaml.Unmarshal(cfgData, &cfg); err != nil {
		return fmt.Errorf("parse config: %w", err)
	}

	// Override milestone from flag if set
	milestoneFlag, _ := cmd.Flags().GetString("milestone")
	defaultMilestone := cfg.DefaultMilestone
	if milestoneFlag != "" {
		defaultMilestone = milestoneFlag
	}

	// Read all stdin
	input, err := io.ReadAll(cmd.InOrStdin())
	if err != nil {
		return fmt.Errorf("read stdin: %w", err)
	}

	// Split into individual issue blocks
	blocks := issue.SplitBatch(input)
	if len(blocks) == 0 {
		return fmt.Errorf("no issues found in input")
	}

	// Parse and create each issue
	now := time.Now()
	var created []*issue.Issue
	for _, block := range blocks {
		iss, err := issue.Parse(block)
		if err != nil {
			return fmt.Errorf("parse issue: %w", err)
		}

		id, err := uuid.NewV7()
		if err != nil {
			return fmt.Errorf("generate UUID: %w", err)
		}
		iss.ID = id
		iss.Created = now
		iss.Updated = now

		if iss.State == "" {
			iss.State = issue.StatePending
		}
		if iss.Milestone == "" {
			iss.Milestone = defaultMilestone
		}

		created = append(created, iss)
	}

	// Wire sequential dependencies for batch creates
	for i := 0; i < len(created)-1; i++ {
		created[i].Blocks = append(created[i].Blocks, created[i+1].ID)
		created[i+1].BlockedBy = append(created[i+1].BlockedBy, created[i].ID)
	}

	// Write all issues to storage
	for _, iss := range created {
		data, err := issue.Marshal(iss)
		if err != nil {
			return fmt.Errorf("marshal issue %s: %w", iss.ID, err)
		}
		refPath := "refs/chain/_/issues/" + iss.ID.String()
		if err := app.Store.WriteEntity(refPath, "issue.md", data, "Create issue: "+iss.Title); err != nil {
			return fmt.Errorf("write issue %s: %w", iss.ID, err)
		}
	}

	// Output results
	// --format is a persistent flag on the root command, not a local flag.
	// Must use Root().PersistentFlags() to find inherited persistent flags.
	format, _ := cmd.Root().PersistentFlags().GetString("format")
	if format == "json" {
		enc := json.NewEncoder(cmd.OutOrStdout())
		enc.SetIndent("", "  ")
		return enc.Encode(created)
	}

	// Human-readable output
	if len(created) == 1 {
		iss := created[0]
		fmt.Fprintf(cmd.OutOrStdout(), "Created %s: %s\n", iss.ID.String()[:8], iss.Title)
	} else {
		fmt.Fprintf(cmd.OutOrStdout(), "Created %d issues (chained sequentially):\n", len(created))
		for i, iss := range created {
			fmt.Fprintf(cmd.OutOrStdout(), "  %s  %s\n", iss.ID.String()[:8], iss.Title)
			if i < len(created)-1 {
				fmt.Fprintf(cmd.OutOrStdout(), "   ↓\n")
			}
		}
	}

	return nil
}
